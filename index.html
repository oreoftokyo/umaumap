function render() { const list = document.getElementById("list"); // エリア順 → 店名順 data.sort((a, b) => { if (a.area !== b.area) { return a.area.localeCompare(b.area, 'ja'); } return a.name.localeCompare(b.name, 'ja'); }); list.innerHTML = ""; let currentArea = ""; data.forEach(d => { // エリア見出し if (d.area !== currentArea) { currentArea = d.area; const h = document.createElement("h3"); h.textContent = currentArea; list.appendChild(h); } const div = document.createElement("div"); div.className = "card"; div.innerHTML = ` ${d.name}
地図で見る `; list.appendChild(div); }); }
